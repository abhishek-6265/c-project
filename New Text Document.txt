#include <stdio.h>

struct product
{
    char name[30];
    float price;
    int stock;
};

int main()
{
    struct product p[50];
    int n = 0, choice, i, sno, qty;
    float amount;

    while(1)
    {
        printf("\n1. Add Product");
        printf("\n2. View All Products");
        printf("\n3. Bill One Item");
        printf("\n4. Exit");

        printf("\nEnter your choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
                printf("Enter product name: ");
                scanf("%s", p[n].name);

                printf("Enter price: ");
                scanf("%f", &p[n].price);

                printf("Enter stock: ");
                scanf("%d", &p[n].stock);

                n++;

                printf("Product added successfully!\n");
                break;

            case 2:
                printf("\nS.No Name Price Stock\n");

                for(i = 0; i < n; i++)
                {
                    printf("%d  %s  %.2f  %d\n",
                           i+1, p[i].name,
                           p[i].price, p[i].stock);
                }
                break;

            case 3:
                printf("Enter product S.No: ");
                scanf("%d", &sno);

                if(sno < 1 || sno > n)
                {
                    printf("Invalid S.No!\n");
                    break;
                }

                printf("Enter quantity: ");
                scanf("%d", &qty);

                if(qty <= 0 || qty > p[sno-1].stock)
                {
                    printf("Invalid quantity or insufficient stock!\n");
                    break;
                }

                amount = p[sno-1].price * qty;

                p[sno-1].stock -= qty;

                printf("\n----- BILL -----\n");
                printf("Product: %s\n", p[sno-1].name);
                printf("Price: %.2f\n", p[sno-1].price);
                printf("Quantity: %d\n", qty);
                printf("Total Amount: %.2f\n", amount);
                printf("---------------\n");

                break;

            case 4:
                return 0;

            default:
                printf("Invalid choice!\n");
        }
    }

    return 0;
}
